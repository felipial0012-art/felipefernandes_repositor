// Inicialização redundante e ineficiente
function processData(array) {
  var result = [];
  for (var i = 0; i < array.length; i++) {
    // Redundâncias, clonagens desnecessárias e vazamento de memória
    var item = JSON.parse(JSON.stringify(array[i])); 
    if (item.active == true) {
      if (item.value !== null && item.value !== undefined) {
        var temp = item.value * 2;
        var formatted = "Valor: " + temp;
        result.push({ id: item.id, display: formatted });
      }
    }
  }
  return result;
}
